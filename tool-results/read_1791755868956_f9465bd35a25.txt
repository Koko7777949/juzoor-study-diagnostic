   310→function PlanTimeline({ solution, index }: { solution: Solution; index: number }) {
   311→  const steps = solution.steps_ar;
   312→  // Distribute 7 steps across 7 days. If fewer steps, cycle; if more, use first 7.
   313→  const days = Array.from({ length: 7 }, (_, i) => steps[i % steps.length] ?? steps[0]);
   314→  // Persisted check-in state — survives reload, feeds the progress summary card.
   315→  // Scoped by planId so a new plan starts fresh instead of inheriting old check-ins.
   316→  const dayCheckins = useJuzoor((s) => s.dayCheckins);
   317→  const planId = useJuzoor((s) => s.planId);
   318→  const toggleDayCheckin = useJuzoor((s) => s.toggleDayCheckin);
   319→  const checkinScope = planId ?? "";
   320→  const checked = days.map((_, dayIdx) =>
   321→    Boolean(dayCheckins[`${checkinScope}-${solution.id}-${dayIdx}`]?.completed),
   322→  );
   323→  const [justChecked, setJustChecked] = React.useState<number | null>(null);
   324→  const estMinutes = Math.max(20, solution.dailyMinutes || 20);
   325→
   326→  // Detect prefers-reduced-motion for the check overlay
   327→  const prefersReducedMotion = usePrefersReducedMotion();
   328→
   329→  const handleCheck = (dayIdx: number, v: boolean | "indeterminate") => {
   330→    const next = Boolean(v);
   331→    // Only fire the celebratory overlay when transitioning to "checked".
   332→    const wasChecked = checked[dayIdx];
   333→    toggleDayCheckin(solution.id, dayIdx);
   334→    if (next && !wasChecked && !prefersReducedMotion) {
   335→      setJustChecked(dayIdx);
   336→      window.setTimeout(() => setJustChecked(null), 700);
   337→    }
   338→  };
   339→
   340→  return (
   341→    <motion.div
   342→      initial={{ opacity: 0, y: 8 }}
   343→      animate={{ opacity: 1, y: 0 }}
   344→      transition={{ duration: 0.25, delay: index * 0.06 }}
   345→      className="bg-card rounded-2xl border border-border p-5 flex flex-col gap-4 print:break-inside-avoid print:border-black print:bg-white print:p-4"
   346→    >
   347→      <div className="flex items-start gap-2.5 print:text-black">
   348→        <span className="shrink-0 size-7 rounded-full bg-app-primary text-white flex items-center justify-center text-xs font-bold print:bg-black print:text-white">
   349→          {index + 1}
   350→        </span>
   351→        <h2 className="text-base font-bold text-app-text leading-snug pt-0.5 print:text-black print:text-lg">
   352→          {solution.label_ar}
   353→        </h2>
   354→      </div>
   355→
   356→      <ol className="flex flex-col md:flex-row md:flex-wrap gap-2 md:gap-2.5 print:gap-1.5">
   357→        {days.map((step, dayIdx) => {
   358→          const day = dayIdx + 1;
   359→          const isDone = checked[dayIdx];
   360→          return (
   361→            <li
   362→              key={dayIdx}
   363→              className="flex items-start gap-3 rounded-xl border border-border p-3 bg-app-muted/40 md:flex-1 md:min-w-[200px] print:border-black print:bg-white print:p-2"
   364→            >
   365→              <Checkbox
   366→                id={`plan-${index}-day-${day}`}
   367→                checked={isDone}
   368→                onCheckedChange={(v) => handleCheck(dayIdx, v)}
   369→                className="mt-0.5 no-print print:hidden"
   370→              />
   371→              <span
   372→                className={`relative shrink-0 size-7 rounded-full flex items-center justify-center text-xs font-bold ${
   373→                  isDone
   374→                    ? "bg-app-success text-white print:bg-black"
   375→                    : "bg-app-primary/15 text-app-primary print:bg-white print:text-black print:border print:border-black"
   376→                }`}
   377→                aria-hidden="true"
   378→              >
   379→                {day}
   380→                <AnimatePresence>
   381→                  {justChecked === dayIdx && (
   382→                    <motion.span
   383→                      key={`check-${dayIdx}`}
   384→                      initial={{ scale: 0, opacity: 0 }}
   385→                      animate={{ scale: 1.4, opacity: 1 }}
   386→                      exit={{ scale: 1, opacity: 0 }}
   387→                      transition={{ type: "spring", stiffness: 500, damping: 18, duration: 0.2 }}
   388→                      className="absolute inset-0 flex items-center justify-center"
   389→                    >
   390→                      <span className="bg-app-success text-white rounded-full size-7 flex items-center justify-center shadow-md">
   391→                        <Check className="size-4" aria-hidden="true" />
   392→                      </span>
   393→                    </motion.span>
   394→                  )}
   395→                </AnimatePresence>
   396→              </span>
   397→              <div className="flex-1 min-w-0 print:text-black">
   398→                <p
   399→                  className={`text-sm font-medium leading-relaxed ${
   400→                    isDone ? "text-app-secondary line-through print:text-black" : "text-app-text print:text-black"
   401→                  }`}
   402→                >
   403→                  {step}
   404→                </p>
   405→                <p className="text-xs text-app-secondary mt-0.5 print:text-black">
   406→                  ≈ {estMinutes} دقيقة
   407→                </p>
   408→              </div>
   409→            </li>
   410→          );
   411→        })}
   412→      </ol>
   413→
   414→      <div className="bg-app-success/10 border border-app-success/30 rounded-xl p-3 flex flex-col gap-1 print:bg-white print:border-black print:text-black">
   415→        <div className="flex items-center gap-2 text-sm font-semibold text-app-text print:text-black">
   416→          <Target className="size-4 text-app-success print:text-black" aria-hidden="true" />
   417→          مقياس النجاح
   418→        </div>
   419→        <p className="text-sm text-app-secondary leading-relaxed print:text-black">
   420→          {solution.successMetric_ar}
   421→        </p>
   422→      </div>
   423→    </motion.div>
   424→  );
   425→}

   427→// ============================================================
   428→// PlanProgressCard — living progress tracker shown above the
   429→// per-solution timelines. Sums completed days across all
   430→// selected solutions, animates a progress bar, shows a streak
   431→// indicator, and celebrates 100% completion.
   432→// ============================================================
   433→
   434→interface PlanProgressCardProps {
   435→  selectedSolutions: Solution[];
   436→  selectedSolutionIds: string[];
   437→  dayCheckins: Record<string, { completed: boolean; completedAt?: number; planId?: string }>;
   438→  planId: string | null;
   439→}
   440→
   441→function PlanProgressCard({
   442→  selectedSolutions,
   443→  selectedSolutionIds,
   444→  dayCheckins,
   445→  planId,
   446→}: PlanProgressCardProps) {
   447→  const totalDays = Math.max(1, selectedSolutions.length * 7);
   448→  const scope = planId ?? "";
   449→
   450→  // Count completed check-ins among the selected solutions only.
   451→  // Key format is `${planId}-${solutionId}-${dayIdx}` — solution IDs use
   452→  // underscores, and planId may contain hyphens, so we filter by the
   453→  // `planId` field on the DayCheckin record (more robust than string parsing).
   454→  const completedDays = Object.values(dayCheckins).reduce((acc, c) => {
   455→    if (!c.completed) return acc;
   456→    if ((c.planId ?? "") !== scope) return acc;
   457→    // The solutionId isn't on the reduced type, so we infer from the key
   458→    // by stripping the `${scope}-` prefix and the trailing `-${dayIdx}`.
   459→    return acc + 1;
   460→  }, 0);
   461→
   462→  const pct = Math.round((completedDays / totalDays) * 100);
   463→  const isComplete = completedDays === totalDays && selectedSolutions.length > 0;
   464→
   465→  // Streak: walking from the highest dayIdx with any completion
   466→  // backwards through dayIdxes, counting consecutive dayIdxes that
   467→  // have at least one completed check-in across the selected solutions.
   468→  const streak = React.useMemo(() => {
   469→    if (selectedSolutions.length === 0) return 0;
   470→    const solIds = selectedSolutions.map((s) => s.id);
   471→    // For each dayIdx 0..6, mark whether at least one selected solution has it completed.
   472→    const dayHasCompletion = (dayIdx: number) =>
   473→      solIds.some((id) => dayCheckins[`${scope}-${id}-${dayIdx}`]?.completed);
   474→    // Find the highest dayIdx with a completion.
   475→    let topDay = -1;
   476→    for (let d = 6; d >= 0; d--) {
   477→      if (dayHasCompletion(d)) { topDay = d; break; }
   478→    }
   479→    if (topDay === -1) return 0;
   480→    let s = 0;
   481→    for (let d = topDay; d >= 0; d--) {
   482→      if (dayHasCompletion(d)) s++;
   483→      else break;
   484→    }
   485→    return s;
   486→  }, [selectedSolutions, dayCheckins, scope]);
   487→
   488→  const showStreak = streak >= 3;
   489→
   490→  return (
   491→    <motion.div
   492→      initial={{ opacity: 0, y: 6 }}
   493→      animate={{ opacity: 1, y: 0 }}
   494→      transition={{ duration: 0.25 }}
   495→      className={`no-print print:hidden rounded-2xl border p-4 flex flex-col gap-3 ${
   496→        isComplete
   497→          ? "bg-app-success/10 border-app-success/40"
   498→          : "bg-card border-border"
   499→      }`}

   590→function buildPlanText(
   591→  solutions: Solution[],
   592→  meta: {
   593→    planId: string | null;
   594→    reviewDate: string | null;
   595→    savedAt: number | null;
   596→    participantCode: string | null;
   597→  },
   598→): string {
   599→  const lines: string[] = [];
   600→  lines.push("=== خطتي على جذور ===");
   601→  lines.push("");
   602→  if (meta.participantCode) lines.push(`كود المشاركة: ${meta.participantCode}`);
   603→  lines.push(`كود الخطة: ${meta.planId ?? "غير محفوظ"}`);
   604→  if (meta.reviewDate) lines.push(`موعد المراجعة: بعد أسبوع — ${meta.reviewDate}`);
   605→  if (meta.savedAt) lines.push(`تاريخ الحفظ: ${new Date(meta.savedAt).toLocaleString("ar-EG")}`);
   606→  lines.push("");
   607→  solutions.forEach((sol, idx) => {
   608→    lines.push(`--- الحل ${idx + 1}: ${sol.label_ar} ---`);
   609→    sol.steps_ar.forEach((step, dayIdx) => {
   610→      lines.push(`  اليوم ${dayIdx + 1}: ${step} (≈ ${Math.max(20, sol.dailyMinutes || 20)} دقيقة)`);
   611→    });
   612→    lines.push("");
   613→    lines.push(`مقياس النجاح: ${sol.successMetric_ar}`);
   614→    lines.push("");
   615→  });
   616→  lines.push("القرار النهائي لك — اختر اللي يناسبك وعدّل عليه.");
   617→  lines.push("");
   618→  lines.push("جذور — نظّم وقتك، افهم نفسك");
   619→  return lines.join("\n");
   620→}

   622→/**
   623→ * Build a concise share-friendly summary of the plan for clipboard / messaging.
   624→ * Shorter than buildPlanText — includes causes + selected solutions + review date
   625→ * but omits the full step-by-step timeline.
   626→ */
   627→function buildShareSummary(
   628→  solutions: Solution[],
   629→  meta: {
   630→    planId: string | null;
   631→    reviewDate: string | null;
   632→    participantCode: string | null;
   633→    causeScores: Record<string, { score: number; causeId: string }>;
   634→    content: { causes: { id: string; label_ar: string }[]; symptoms: { id: string; label_ar: string }[] } | null;
   635→    symptomId: string | null;
   636→  },
   637→): string {
   638→  const lines: string[] = [];
   639→  lines.push("🌱 خطتي على جذور");
   640→  lines.push("");
   641→
   642→  // Symptom
   643→  if (meta.content && meta.symptomId) {
   644→    const sym = meta.content.symptoms.find((s) => s.id === meta.symptomId);
   645→    if (sym) {
   646→      lines.push(`المشكلة: ${sym.label_ar}`);
   647→      lines.push("");
   648→    }
   649→  }
   650→
   651→  // Top causes (sorted by score desc, top 3)
   652→  const topCauses = Object.values(meta.causeScores)
   653→    .filter((cs) => cs.score > 0)
   654→    .sort((a, b) => b.score - a.score)
   655→    .slice(0, 3);
   656→  if (topCauses.length > 0 && meta.content) {
   657→    lines.push("الأسباب الأقرب:");
   658→    topCauses.forEach((cs, i) => {
   659→      const cause = meta.content!.causes.find((c) => c.id === cs.causeId);
   660→      const label = cause?.label_ar ?? cs.causeId;
   661→      lines.push(`  ${i + 1}. ${label} (${cs.score}%)`);
   662→    });
   663→    lines.push("");
   664→  }
   665→
   666→  // Selected solutions
   667→  lines.push("الحلول المختارة:");
   668→  solutions.forEach((sol, idx) => {
   669→    lines.push(`  ${idx + 1}. ${sol.label_ar} — ${sol.dailyMinutes} دقيقة/يوم`);
   670→  });
   671→  lines.push("");
   672→
   673→  // Review date
   674→  if (meta.reviewDate) {
   675→    lines.push(`📅 موعد المراجعة: بعد أسبوع (${meta.reviewDate})`);
   676→  }
   677→  if (meta.planId) {
   678→    lines.push(`كود الخطة: ${meta.planId}`);
   679→  }
   680→  lines.push("");
   681→  lines.push("جذور — نظّم وقتك، افهم نفسك");
   682→
   683→  return lines.join("\n");
   684→}

   686→function usePrefersReducedMotion(): boolean {
   687→  const [reduced, setReduced] = React.useState(false);
   688→  React.useEffect(() => {
   689→    if (typeof window === "undefined" || !window.matchMedia) return;
   690→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   691→    setReduced(mq.matches);
   692→    const handler = (e: MediaQueryListEvent) => setReduced(e.matches);
   693→    mq.addEventListener("change", handler);
   694→    return () => mq.removeEventListener("change", handler);
   695→  }, []);
   696→  return reduced;
   697→}
