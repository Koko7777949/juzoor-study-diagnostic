    40→const DEFAULT_WEIGHTS = { wE: 35, wA: 25, wS: 15, wF: 25 };

    42→const MAX_SELECTED = 3;

    44→/** Returns the CSS variable name for a score bucket. */
    45→function scoreBucket(score: number): { colorVar: string; label: string } {
    46→  if (score >= 80) return { colorVar: "var(--app-success)", label: "ممتاز" };
    47→  if (score >= 60) return { colorVar: "var(--app-primary)", label: "جيد" };
    48→  if (score >= 40) return { colorVar: "var(--confidence-medium)", label: "متوسط" };
    49→  return { colorVar: "var(--confidence-low)", label: "منخفض" };
    50→}

    52→export function SolutionsScreen() {
    53→  const content = useJuzoor((s) => s.content);
    54→  const solutionScores = useJuzoor((s) => s.solutionScores);
    55→  const selectedSolutionIds = useJuzoor((s) => s.selectedSolutionIds);
    56→  const weights = useJuzoor((s) => s.weights);
    57→  const setWeights = useJuzoor((s) => s.setWeights);
    58→  const toggleSolution = useJuzoor((s) => s.toggleSolution);
    59→  const goToScenarios = useJuzoor((s) => s.goToScenarios);
    60→
    61→  // Expansion state — multiple solutions can be open simultaneously so the
    62→  // user can compare details side-by-side. Set<id> for O(1) lookup.
    63→  const [expandedIds, setExpandedIds] = React.useState<Set<string>>(
    64→    () => new Set<string>(),
    65→  );
    66→
    67→  const toggleExpand = React.useCallback((id: string) => {
    68→    setExpandedIds((prev) => {
    69→      const next = new Set(prev);
    70→      if (next.has(id)) next.delete(id);
    71→      else next.add(id);
    72→      return next;
    73→    });
    74→  }, []);
    75→
    76→  if (!content) return null;
    77→
    78→  // ============================================================
    79→  // Empty state (defensive — engine always emits 3+ solutions when
    80→  // reaching this screen, but if data is missing we don't want to crash)
    81→  // ============================================================
    82→  if (solutionScores.length === 0) {
    83→    return <EmptySolutionsState />;
    84→  }
    85→
    86→  const weightValues = {
    87→    wE: Math.round(weights.wE * 100),
    88→    wA: Math.round(weights.wA * 100),
    89→    wS: Math.round(weights.wS * 100),
    90→    wF: Math.round(weights.wF * 100),
    91→  };
    92→
    93→  const handleSliderChange = (key: "wE" | "wA" | "wS" | "wF", val: number) => {
    94→    const next = { ...weightValues, [key]: val };
    95→    setWeights({
    96→      wE: next.wE / 100,
    97→      wA: next.wA / 100,
    98→      wS: next.wS / 100,
    99→      wF: next.wF / 100,
   100→    });
   101→  };
   102→
   103→  const resetWeights = () => {
   104→    setWeights({
   105→      ...DEFAULT_WEIGHTS,
   106→      wE: DEFAULT_WEIGHTS.wE / 100,
   107→      wA: DEFAULT_WEIGHTS.wA / 100,
   108→      wS: DEFAULT_WEIGHTS.wS / 100,
   109→      wF: DEFAULT_WEIGHTS.wF / 100,
   110→    });
   111→    toast.success("تمت إعادة ضبط الأوزان للقيم الافتراضية");
   112→  };
   113→
   114→  const handleRowToggle = (id: string) => {
   115→    const isSelected = selectedSolutionIds.includes(id);
   116→    if (!isSelected && selectedSolutionIds.length >= MAX_SELECTED) return;
   117→    toggleSolution(id);
   118→  };
   119→
   120→  return (
   121→    <div className="max-w-md mx-auto px-4 pt-6 pb-12 flex flex-col gap-6">
   122→      <header className="flex flex-col gap-1.5 pt-2">
   123→        <h1 className="text-2xl font-bold text-app-text leading-snug">مقارنة الحلول</h1>
   124→        <p className="text-sm text-app-secondary leading-relaxed">
   125→          نصيحتك بالحلول الأنسب حسب أسبابك ووقتك
   126→        </p>
   127→      </header>
   128→
   129→      {/* Weight sliders — inline (not sticky to avoid overlapping table content) */}
   130→      <div className="bg-app-muted/40 rounded-xl p-3 border border-app-primary/10">
   131→        <WeightSliders
   132→          values={weightValues}
   133→          onChange={handleSliderChange}
   134→          onReset={resetWeights}
   135→        />
   136→        <p className="text-xs text-app-secondary mt-2 flex items-center gap-1.5 px-1">
   137→          <SlidersHorizontal className="size-3 shrink-0 text-app-primary" aria-hidden="true" />
   138→          حرّك الأوزان وشوف الترتيب بيتغير فوراً
   139→        </p>
   140→      </div>
   141→
   142→      <Separator label="جدول المقارنة" />
   143→
   144→      {/* Solution count summary */}
   145→      <div className="flex items-center justify-between text-xs text-app-secondary px-1">
   146→        <span className="flex items-center gap-1.5">
   147→          <ListTree className="size-3.5 shrink-0 text-app-primary" aria-hidden="true" />
   148→          عدد الحلول:{" "}
   149→          <span className="font-bold text-app-text tabular-nums">
   150→            {solutionScores.length}
   151→          </span>
   152→          <span className="text-app-secondary/60" aria-hidden="true">—</span>
   153→          المختار:{" "}
   154→          <span className="font-bold text-app-primary tabular-nums">
   155→            {selectedSolutionIds.length}
   156→          </span>{" "}
   157→          من <span className="tabular-nums">{MAX_SELECTED}</span>
   158→        </span>
   159→      </div>
   160→
   161→      {/* Comparison matrix - desktop table / mobile cards */}
   162→      <div className="md:block hidden">
   163→        <ComparisonTable
   164→          scores={solutionScores}
   165→          solutions={content.solutions}
   166→          selectedIds={selectedSolutionIds}
   167→          onToggle={handleRowToggle}
   168→          expandedIds={expandedIds}
   169→          onToggleExpand={toggleExpand}
   170→          causes={content.causes}
   171→        />
   172→      </div>
   173→      <div className="md:hidden flex flex-col gap-3">
   174→        {solutionScores.map((sc, idx) => {
   175→          const sol = content.solutions.find((s) => s.id === sc.solutionId);
   176→          if (!sol) return null;
   177→          const selected = selectedSolutionIds.includes(sc.solutionId);
   178→          const maxSelected = selectedSolutionIds.length >= MAX_SELECTED && !selected;
   179→          const isTop = idx === 0; // first in sorted list = #1 priority
   180→          return (
   181→            <SolutionCard
   182→              key={sc.solutionId}
   183→              solution={sol}
   184→              score={sc}
   185→              selected={selected}
   186→              maxSelected={maxSelected}
   187→              isTop={isTop}
   188→              onToggle={() => handleRowToggle(sc.solutionId)}
   189→              causes={content.causes}
   190→              expanded={expandedIds.has(sc.solutionId)}
   191→              onToggleExpand={() => toggleExpand(sc.solutionId)}
   192→            />
   193→          );
   194→        })}
   195→      </div>
   196→
   197→      {/* Color legend */}
   198→      <ScoreLegend />
   199→
   200→      <Button
   201→        type="button"
   202→        className="touch-target h-12 text-base bg-app-primary text-white hover:bg-app-primary/90 mt-2"
   203→        onClick={goToScenarios}
   204→        disabled={selectedSolutionIds.length === 0}
   205→      >
   206→        اعرض السيناريوهات
   207→        <ChevronLeft className="size-5" aria-hidden="true" />
   208→      </Button>
   209→    </div>
   210→  );
   211→}

   294→function WeightSliders({
   295→  values,
   296→  onChange,
   297→  onReset,
   298→}: {
   299→  values: { wE: number; wA: number; wS: number; wF: number };
   300→  onChange: (key: "wE" | "wA" | "wS" | "wF", val: number) => void;
   301→  onReset: () => void;
   302→}) {
   303→  const sliders = [
   304→    { key: "wE" as const, label: "الفاعلية" },
   305→    { key: "wA" as const, label: "السهولة" },
   306→    { key: "wS" as const, label: "السرعة" },
   307→    { key: "wF" as const, label: "الملاءمة" },
   308→  ];
   309→
   310→  const sum = values.wE + values.wA + values.wS + values.wF || 1;
   311→
   312→  return (
   313→    <div className="bg-card rounded-2xl border border-border p-5 flex flex-col gap-4">
   314→      <div className="flex items-center justify-between">
   315→        <h2 className="text-base font-semibold text-app-text flex items-center gap-2">
   316→          <SlidersHorizontal className="size-4 text-app-primary" aria-hidden="true" />
   317→          اضبط أولوياتك
   318→        </h2>
   319→        <Button
   320→          type="button"
   321→          variant="ghost"
   322→          size="sm"
   323→          className="touch-target text-xs text-app-secondary hover:text-app-primary px-2"
   324→          onClick={onReset}
   325→        >
   326→          <RotateCcw className="size-3.5" aria-hidden="true" />
   327→          إعادة الضبط
   328→        </Button>
   329→      </div>
   330→
   331→      <div className="flex flex-col gap-3">
   332→        {sliders.map((s) => (
   333→          <div key={s.key} className="flex flex-col gap-1.5">
   334→            <div className="flex items-center justify-between text-sm">
   335→              <Label htmlFor={`slider-${s.key}`} className="text-app-text font-medium">
   336→                {s.label}
   337→              </Label>
   338→              <span className="font-bold text-app-primary tabular-nums">
   339→                {Math.round((values[s.key] / sum) * 100)}%
   340→              </span>
   341→            </div>
   342→            <Slider
   343→              id={`slider-${s.key}`}
   344→              value={[values[s.key]]}
   345→              min={0}
   346→              max={100}
   347→              step={5}
   348→              onValueChange={(v) => onChange(s.key, v[0])}
   349→              aria-label={s.label}
   350→            />
   351→          </div>
   352→        ))}
   353→      </div>
   354→
   355→      <p className="text-xs text-app-secondary leading-relaxed">
   356→        النسب بتتعدّل تلقائياً بحيث يكون مجموعها 100%.
   357→      </p>
   358→    </div>
   359→  );
   360→}

   362→function causeLabelFor(
   363→  score: SolutionScore,
   364→  causes: { id: string; label_ar: string }[],
   365→): string {
   366→  const top = causes.find((c) => c.id === score.causeIds[0]);
   367→  return top?.label_ar ?? score.causeIds[0];
   368→}

   370→// ============================================================
   371→// Comparison table (desktop)
   372→// Uses Radix Collapsible with `asChild` to render as <tbody> so each row
   373→// group (data row + details row) is its own Collapsible unit.
   374→// ============================================================
   375→function ComparisonTable({
   376→  scores,
   377→  solutions,
   378→  selectedIds,
   379→  onToggle,
   380→  expandedIds,
   381→  onToggleExpand,
   382→  causes,
   383→}: {
   384→  scores: SolutionScore[];
   385→  solutions: Solution[];
   386→  selectedIds: string[];
   387→  onToggle: (id: string) => void;
   388→  expandedIds: Set<string>;
   389→  onToggleExpand: (id: string) => void;
   390→  causes: { id: string; label_ar: string }[];
   391→}) {
   392→  return (
   393→    <div className="rounded-2xl border border-border bg-card overflow-hidden">
   394→      <Table>
   395→        <TableHeader>
   396→          <TableRow className="bg-app-muted">
   397→            <TableHead className="text-center text-app-secondary font-semibold text-xs w-10">اختيار</TableHead>
   398→            <TableHead className="text-center text-app-secondary font-semibold text-xs w-12">تفاصيل</TableHead>
   399→            <TableHead className="text-app-secondary font-semibold text-xs">الحل</TableHead>
   400→            <TableHead className="text-center text-app-secondary font-semibold text-xs">الفاعلية</TableHead>
   401→            <TableHead className="text-center text-app-secondary font-semibold text-xs">السهولة</TableHead>
   402→            <TableHead className="text-center text-app-secondary font-semibold text-xs">السرعة</TableHead>
   403→            <TableHead className="text-center text-app-secondary font-semibold text-xs">الملاءمة</TableHead>
   404→            <TableHead className="text-center text-app-secondary font-semibold text-xs">الدرجة</TableHead>
   405→            <TableHead className="text-center text-app-secondary font-semibold text-xs">الأولوية</TableHead>
   406→            <TableHead className="text-center text-app-secondary font-semibold text-xs">دقيقة/يوم</TableHead>
   407→          </TableRow>
   408→        </TableHeader>
   409→        <TableBody>
   410→          {scores.map((sc, idx) => {
   411→            const sol = solutions.find((s) => s.id === sc.solutionId);
   412→            if (!sol) return null;
   413→            const selected = selectedIds.includes(sc.solutionId);
   414→            const disabled = !selected && selectedIds.length >= MAX_SELECTED;
   415→            const bucket = scoreBucket(sc.score);
   416→            const isExpanded = expandedIds.has(sc.solutionId);
   417→            const isTop = idx === 0;
   418→
   419→            return (
   420→              <Collapsible
   421→                key={sc.solutionId}
   422→                asChild
   423→                open={isExpanded}
   424→                onOpenChange={() => onToggleExpand(sc.solutionId)}
   425→              >
   426→                <tbody
   427→                  className={cn(
   428→                    "group scroll-mt-40",
   429→                    selected && "bg-app-primary/5",
   430→                    isTop && "ring-1 ring-inset ring-amber-400/40",
   431→                  )}
   432→                >
   433→                  <TableRow
   434→                    className={cn(
   435→                      "border-b",
   436→                      isTop && "bg-amber-50/40",
   437→                      isExpanded && "border-b-app-primary/30",
   438→                    )}
   439→                  >
   440→                    {/* Checkbox cell */}
   441→                    <TableCell className="text-center">
   442→                      <Checkbox
   443→                        checked={selected}
   444→                        disabled={disabled}
   445→                        onCheckedChange={() => onToggle(sc.solutionId)}
   446→                        onClick={(e) => e.stopPropagation()}
   447→                        aria-label={`اختيار ${sol.label_ar}`}
   448→                      />
   449→                    </TableCell>
   450→
   451→                    {/* "تفاصيل" trigger button cell */}
   452→                    <TableCell className="text-center">
   453→                      <CollapsibleTrigger asChild>
   454→                        <button
   455→                          type="button"
   456→                          className="size-8 rounded-md inline-flex items-center justify-center text-app-secondary hover:text-app-primary hover:bg-app-primary/10 transition-colors"
   457→                          aria-label="تفاصيل الحل"
   458→                          aria-expanded={isExpanded}
   459→                          onClick={(e) => e.stopPropagation()}
   460→                        >
   461→                          <ChevronDown
   462→                            className={cn(
   463→                              "size-4 transition-transform duration-200",
   464→                              isExpanded && "rotate-180",
   465→                            )}
   466→                            aria-hidden="true"
   467→                          />
   468→                        </button>
   469→                      </CollapsibleTrigger>
   470→                    </TableCell>
   471→
   472→                    {/* Solution name cell — also a trigger (per task a) */}
   473→                    <TableCell className="text-app-text font-medium text-xs max-w-[180px]">
   474→                      <CollapsibleTrigger asChild>
   475→                        <button
   476→                          type="button"
   477→                          className="text-start flex items-center gap-1.5 hover:text-app-primary transition-colors"
   478→                          aria-label={`عرض تفاصيل ${sol.label_ar}`}
   479→                          aria-expanded={isExpanded}
   480→                          onClick={(e) => e.stopPropagation()}
   481→                        >
   482→                          {isTop && (
   483→                            <span className="inline-flex items-center gap-0.5 shrink-0 bg-amber-400 text-amber-950 rounded-full px-1.5 py-0.5 text-[10px] font-bold">
   484→                              <Crown className="size-2.5" aria-hidden="true" />
   485→                              الأفضل لك
   486→                            </span>
   487→                          )}
   488→                          <span className="line-clamp-2">{sol.label_ar}</span>
   489→                        </button>
   490→                      </CollapsibleTrigger>
   491→                    </TableCell>
   492→
   493→                    <TableCell className="text-center text-app-text tabular-nums">{sc.E}/5</TableCell>
   494→                    <TableCell className="text-center text-app-text tabular-nums">{sc.A}/5</TableCell>
   495→                    <TableCell className="text-center text-app-text tabular-nums">{sc.S}/5</TableCell>
   496→                    <TableCell className="text-center text-app-text tabular-nums">{sc.fit}/5</TableCell>
   497→
   498→                    {/* Score cell with color bar */}
   499→                    <TableCell className="text-center">
   500→                      <div className="flex items-center gap-1.5 justify-center">
   501→                        <div className="w-10 h-1.5 rounded-full overflow-hidden" style={{ backgroundColor: "var(--app-muted)" }}>
   502→                          <div
   503→                            className="h-full rounded-full"
   504→                            style={{ width: `${sc.score}%`, backgroundColor: bucket.colorVar }}
   505→                          />
   506→                        </div>
   507→                        <span
   508→                          className="font-bold text-xs tabular-nums"
   509→                          style={{ color: bucket.colorVar }}
   510→                        >
   511→                          {sc.score}
   512→                        </span>
   513→                      </div>
   514→                    </TableCell>
   515→
   516→                    {/* Priority cell */}
   517→                    <TableCell className="text-center">
   518→                      <div className="flex items-center justify-center gap-1">
   519→                        {selected && (
   520→                          <ArrowUp
   521→                            className="size-3 text-app-primary"
   522→                            aria-label="مختار"
   523→                            aria-hidden="true"
   524→                          />
   525→                        )}
   526→                        <span className="font-bold text-app-primary tabular-nums">{sc.priority}</span>
   527→                      </div>
   528→                    </TableCell>
   529→
   530→                    <TableCell className="text-center text-app-text tabular-nums">{sol.dailyMinutes}</TableCell>
   531→                  </TableRow>
   532→
   533→                  {/* Details row — spans all 10 columns, hidden when collapsed */}
   534→                  <CollapsibleContent asChild>
   535→                    <TableRow className="data-[state=closed]:hidden">
   536→                      <TableCell colSpan={10} className="bg-app-muted/30">
   537→                        <SolutionDetailPanel
   538→                          solution={sol}
   539→                          score={sc}
   540→                          causes={causes}
   541→                        />
   542→                      </TableCell>
   543→                    </TableRow>
   544→                  </CollapsibleContent>
   545→                </tbody>
   546→              </Collapsible>
   547→            );
   548→          })}
   549→        </TableBody>
   550→      </Table>
   551→    </div>
   552→  );
   553→}

   555→// ============================================================
   556→// Solution detail panel (reusable expansion content)
   557→// Shown inside both the desktop table details row and the mobile card.
   558→// ============================================================
   559→function SolutionDetailPanel({
   560→  solution,
   561→  score,
   562→  causes,
   563→}: {
   564→  solution: Solution;
   565→  score: SolutionScore;
   566→  causes: { id: string; label_ar: string }[];
   567→}) {
   568→  // Lookup cause labels from content.causes
   569→  const causeLabels: string[] = score.causeIds.map((id) => {
   570→    const c = causes.find((cc) => cc.id === id);
   571→    return c?.label_ar ?? id;
   572→  });
   573→
   574→  // Scenarios preview — truncate expected_ar to ~80 chars
   575→  const expected = solution.scenarios?.expected_ar ?? "";
   576→  const truncated =
   577→    expected.length > 80 ? `${expected.slice(0, 80).trimEnd()}…` : expected;
   578→
   579→  // First step + remaining count
   580→  const firstStep = solution.steps_ar?.[0] ?? "";
   581→  const remainingSteps = Math.max(0, (solution.steps_ar?.length ?? 0) - 1);
   582→
   583→  return (
   584→    <motion.div
   585→      initial={{ opacity: 0 }}
   586→      animate={{ opacity: 1 }}
   587→      transition={{ duration: 0.2 }}
   588→      className="flex flex-col gap-3 py-3"
   589→    >
   590→      {/* CauseIds row */}
   591→      <div className="flex flex-wrap items-center gap-1.5">
   592→        <span className="text-[10px] font-semibold text-app-secondary uppercase tracking-wider">
   593→          الأسباب:
   594→        </span>
   595→        {causeLabels.map((label, i) => (
   596→          <Badge
   597→            key={`${label}-${i}`}
   598→            variant="outline"
   599→            className="text-app-primary border-app-primary/30 bg-app-primary/5 gap-1"
   600→          >
   601→            <ListTree className="size-2.5" aria-hidden="true" />
   602→            {label}
   603→          </Badge>
   604→        ))}
   605→      </div>
   606→
   607→      {/* Scenarios preview */}
   608→      {truncated && (
   609→        <div className="flex flex-col gap-1">
   610→          <span className="text-[10px] font-semibold text-app-secondary uppercase tracking-wider">
   611→            السيناريو المتوقع:
   612→          </span>
   613→          <p className="text-xs text-app-text leading-relaxed">{truncated}</p>
   614→          <p className="text-[10px] text-app-secondary/70 flex items-center gap-1">
   615→            <ChevronLeft className="size-2.5" aria-hidden="true" />
   616→            اعرض كل السيناريوهات في الصفحة التالية
   617→          </p>
   618→        </div>
   619→      )}
   620→
   621→      {/* First step */}
   622→      {firstStep && (
   623→        <div className="flex flex-col gap-1">
   624→          <span className="text-[10px] font-semibold text-app-secondary uppercase tracking-wider">
   625→            أول خطوة:
   626→          </span>
   627→          <p className="text-xs text-app-text leading-relaxed ps-3 border-s-2 border-app-primary/40">
   628→            {firstStep}
   629→          </p>
   630→          {remainingSteps > 0 && (
   631→            <Badge
   632→              variant="outline"
   633→              className="self-start text-app-secondary border-app-secondary/30 bg-app-muted/40 mt-1"
   634→            >
   635→              + {remainingSteps} خطوات تانية
   636→            </Badge>
   637→          )}
   638→        </div>
   639→      )}
   640→
   641→      {/* Success metric in a small green-accent card */}
   642→      {solution.successMetric_ar && (
   643→        <div className="bg-app-success/10 border border-app-success/30 rounded-lg p-2.5 flex items-start gap-2">
   644→          <Check className="size-3.5 text-app-success shrink-0 mt-0.5" aria-hidden="true" />
   645→          <div className="flex flex-col gap-0.5">
   646→            <span className="text-[10px] font-semibold text-app-success uppercase tracking-wider">
   647→              مؤشر النجاح:
   648→            </span>
   649→            <p className="text-xs text-app-text leading-relaxed">
   650→              {solution.successMetric_ar}
   651→            </p>
   652→          </div>
   653→        </div>
   654→      )}
   655→    </motion.div>
   656→  );
   657→}
