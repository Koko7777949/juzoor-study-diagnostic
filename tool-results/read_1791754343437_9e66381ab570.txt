    44→    44→/** Returns the CSS variable name for a score bucket. */
    45→    45→function scoreBucket(score: number): { colorVar: string; label: string } {
    46→    46→  if (score >= 80) return { colorVar: "var(--app-success)", label: "ممتاز" };
    47→    47→  if (score >= 60) return { colorVar: "var(--app-primary)", label: "جيد" };
    48→    48→  if (score >= 40) return { colorVar: "var(--confidence-medium)", label: "متوسط" };
    49→    49→  return { colorVar: "var(--confidence-low)", label: "منخفض" };
    50→    50→}
