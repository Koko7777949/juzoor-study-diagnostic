    57→const DEFAULT_WEIGHTS = { wE: 35, wA: 25, wS: 15, wF: 25 };

    61→/** Returns the CSS variable name for a score bucket. */
    62→function scoreBucket(score: number): { colorVar: string; label: string } {
    63→  if (score >= 80) return { colorVar: "var(--app-success)", label: "ممتاز" };
    64→  if (score >= 60) return { colorVar: "var(--app-primary)", label: "جيد" };
    65→  if (score >= 40) return { colorVar: "var(--confidence-medium)", label: "متوسط" };
    66→  return { colorVar: "var(--confidence-low)", label: "منخفض" };
    67→}

    69→export function SolutionsScreen() {
    70→  const content = useJuzoor((s) => s.content);
    71→  const solutionScores = useJuzoor((s) => s.solutionScores);
    72→  const selectedSolutionIds = useJuzoor((s) => s.selectedSolutionIds);
    73→  const weights = useJuzoor((s) => s.weights);
    74→  const setWeights = useJuzoor((s) => s.setWeights);
    75→  const toggleSolution = useJuzoor((s) => s.toggleSolution);
    76→  const goToScenarios = useJuzoor((s) => s.goToScenarios);
    77→  const profile = useJuzoor((s) => s.engine.profile);
    78→
    79→  // Expansion state — multiple solutions can be open simultaneously so the
    80→  // user can compare details side-by-side. Set<id> for O(1) lookup.
    81→  const [expandedIds, setExpandedIds] = React.useState<Set<string>>(
    82→    () => new Set<string>(),
    83→  );
    84→
    85→  // Side-by-side comparison dialog state
    86→  const [compareOpen, setCompareOpen] = React.useState(false);
    87→
    88→  const toggleExpand = React.useCallback((id: string) => {
    89→    setExpandedIds((prev) => {
    90→      const next = new Set(prev);
    91→      if (next.has(id)) next.delete(id);
    92→      else next.add(id);
    93→      return next;
    94→    });
    95→  }, []);
    96→
    97→  if (!content) return null;
    98→
    99→  // ============================================================
   100→  // Empty state (defensive — engine always emits 3+ solutions when
   101→  // reaching this screen, but if data is missing we don't want to crash)
   102→  // ============================================================
   103→  if (solutionScores.length === 0) {
   104→    return <EmptySolutionsState />;
   105→  }
   106→
   107→  const weightValues = {
   108→    wE: Math.round(weights.wE * 100),
   109→    wA: Math.round(weights.wA * 100),
   110→    wS: Math.round(weights.wS * 100),
   111→    wF: Math.round(weights.wF * 100),
   112→  };
   113→
   114→  const handleSliderChange = (key: "wE" | "wA" | "wS" | "wF", val: number) => {
   115→    const next = { ...weightValues, [key]: val };
   116→    setWeights({
   117→      wE: next.wE / 100,
   118→      wA: next.wA / 100,
   119→      wS: next.wS / 100,
   120→      wF: next.wF / 100,
   121→    });
   122→  };
   123→
   124→  const resetWeights = () => {
   125→    setWeights({
   126→      ...DEFAULT_WEIGHTS,
   127→      wE: DEFAULT_WEIGHTS.wE / 100,
   128→      wA: DEFAULT_WEIGHTS.wA / 100,
   129→      wS: DEFAULT_WEIGHTS.wS / 100,
   130→      wF: DEFAULT_WEIGHTS.wF / 100,
   131→    });
   132→    toast.success("تمت إعادة ضبط الأوزان للقيم الافتراضية");
   133→  };
   134→
   135→  const handleRowToggle = (id: string) => {
   136→    const isSelected = selectedSolutionIds.includes(id);
   137→    if (!isSelected && selectedSolutionIds.length >= MAX_SELECTED) return;
   138→    toggleSolution(id);
   139→  };
   140→
   141→  return (
   142→    <div className="max-w-md mx-auto px-4 pt-6 pb-12 flex flex-col gap-6">
   143→      <header className="flex flex-col gap-1.5 pt-2">
   144→        <h1 className="text-2xl font-bold text-app-text leading-snug">مقارنة الحلول</h1>
   145→        <p className="text-sm text-app-secondary leading-relaxed">
   146→          نصيحتك بالحلول الأنسب حسب أسبابك ووقتك
   147→        </p>
   148→      </header>
   149→
   150→      {/* Weight sliders — inline (not sticky to avoid overlapping table content) */}
   151→      <div className="bg-app-muted/40 rounded-xl p-3 border border-app-primary/10">
   152→        <WeightSliders
   153→          values={weightValues}
   154→          onChange={handleSliderChange}
   155→          onReset={resetWeights}
   156→        />
   157→        <p className="text-xs text-app-secondary mt-2 flex items-center gap-1.5 px-1">
   158→          <SlidersHorizontal className="size-3 shrink-0 text-app-primary" aria-hidden="true" />
   159→          حرّك الأوزان وشوف الترتيب بيتغير فوراً
   160→        </p>
   161→      </div>
   162→
   163→      {/* Side-by-side comparison trigger */}
   164→      <div className="flex justify-center">
   165→        <Button
   166→          type="button"
   167→          variant="outline"
   168→          className="touch-target h-11 border-app-primary/40 text-app-primary hover:bg-app-primary/5 hover:text-app-primary px-5 gap-2"
   169→          onClick={() => setCompareOpen(true)}
   170→          disabled={solutionScores.length < 2}
   171→        >
   172→          <GitCompare className="size-4" aria-hidden="true" />
   173→          قارن حلوين
   174→        </Button>
   175→      </div>
   176→
   177→      <Separator label="جدول المقارنة" />
   178→
   179→      {/* Solution count summary */}
   180→      <div className="flex items-center justify-between text-xs text-app-secondary px-1">
   181→        <span className="flex items-center gap-1.5">
   182→          <ListTree className="size-3.5 shrink-0 text-app-primary" aria-hidden="true" />
   183→          عدد الحلول:{" "}
   184→          <span className="font-bold text-app-text tabular-nums">
   185→            {solutionScores.length}
   186→          </span>
   187→          <span className="text-app-secondary/60" aria-hidden="true">—</span>
   188→          المختار:{" "}
   189→          <span className="font-bold text-app-primary tabular-nums">
   190→            {selectedSolutionIds.length}
   191→          </span>{" "}
   192→          من <span className="tabular-nums">{MAX_SELECTED}</span>
   193→        </span>
   194→      </div>
   195→
   196→      {/* Comparison matrix - desktop table / mobile cards */}
   197→      <div className="md:block hidden">
   198→        <ComparisonTable
   199→          scores={solutionScores}
   200→          solutions={content.solutions}
   201→          selectedIds={selectedSolutionIds}
   202→          onToggle={handleRowToggle}
   203→          expandedIds={expandedIds}
   204→          onToggleExpand={toggleExpand}
   205→          causes={content.causes}
   206→        />
   207→      </div>
   208→      <div className="md:hidden flex flex-col gap-3">
   209→        {solutionScores.map((sc, idx) => {
   210→          const sol = content.solutions.find((s) => s.id === sc.solutionId);
   211→          if (!sol) return null;
   212→          const selected = selectedSolutionIds.includes(sc.solutionId);
   213→          const maxSelected = selectedSolutionIds.length >= MAX_SELECTED && !selected;
   214→          const isTop = idx === 0; // first in sorted list = #1 priority
   215→          return (
   216→            <SolutionCard
   217→              key={sc.solutionId}
   218→              solution={sol}
   219→              score={sc}
   220→              selected={selected}
   221→              maxSelected={maxSelected}
   222→              isTop={isTop}
   223→              onToggle={() => handleRowToggle(sc.solutionId)}
   224→              causes={content.causes}
   225→              expanded={expandedIds.has(sc.solutionId)}
   226→              onToggleExpand={() => toggleExpand(sc.solutionId)}
   227→            />
   228→          );
   229→        })}
   230→      </div>
   231→
   232→      {/* Color legend */}
   233→      <ScoreLegend />
   234→
   235→      <Button
   236→        type="button"
   237→        className="touch-target h-12 text-base bg-app-primary text-white hover:bg-app-primary/90 mt-2"
   238→        onClick={goToScenarios}
   239→        disabled={selectedSolutionIds.length === 0}
   240→      >
   241→        اعرض السيناريوهات
   242→        <ChevronLeft className="size-5" aria-hidden="true" />
   243→      </Button>
   244→
   245→      {/* Side-by-side comparison dialog */}
   246→      <SolutionComparisonDialog
   247→        open={compareOpen}
   248→        onOpenChange={setCompareOpen}
   249→        scores={solutionScores}
   250→        solutions={content.solutions}
   251→        causes={content.causes}
   252→        profile={profile}
   253→        onGoToScenarios={goToScenarios}
   254→      />
   255→    </div>
   256→  );
   257→}

   340→function WeightSliders({
   341→  values,
   342→  onChange,
   343→  onReset,
   344→}: {
   345→  values: { wE: number; wA: number; wS: number; wF: number };
   346→  onChange: (key: "wE" | "wA" | "wS" | "wF", val: number) => void;
   347→  onReset: () => void;
   348→}) {
   349→  const sliders = [
   350→    { key: "wE" as const, label: "الفاعلية" },
   351→    { key: "wA" as const, label: "السهولة" },
   352→    { key: "wS" as const, label: "السرعة" },
   353→    { key: "wF" as const, label: "الملاءمة" },
   354→  ];
   355→
   356→  const sum = values.wE + values.wA + values.wS + values.wF || 1;
   357→
   358→  return (
   359→    <div className="bg-card rounded-2xl border border-border p-5 flex flex-col gap-4">
   360→      <div className="flex items-center justify-between">
   361→        <h2 className="text-base font-semibold text-app-text flex items-center gap-2">
   362→          <SlidersHorizontal className="size-4 text-app-primary" aria-hidden="true" />
   363→          اضبط أولوياتك
   364→        </h2>
   365→        <Button
   366→          type="button"
   367→          variant="ghost"
   368→          size="sm"
   369→          className="touch-target text-xs text-app-secondary hover:text-app-primary px-2"
   370→          onClick={onReset}
   371→        >
   372→          <RotateCcw className="size-3.5" aria-hidden="true" />
   373→          إعادة الضبط
   374→        </Button>
   375→      </div>
   376→
   377→      <div className="flex flex-col gap-3">
   378→        {sliders.map((s) => (
   379→          <div key={s.key} className="flex flex-col gap-1.5">
   380→            <div className="flex items-center justify-between text-sm">
   381→              <Label htmlFor={`slider-${s.key}`} className="text-app-text font-medium">
   382→                {s.label}
   383→              </Label>
   384→              <span className="font-bold text-app-primary tabular-nums">
   385→                {Math.round((values[s.key] / sum) * 100)}%
   386→              </span>
   387→            </div>
   388→            <Slider
   389→              id={`slider-${s.key}`}
   390→              value={[values[s.key]]}
   391→              min={0}
   392→              max={100}
   393→              step={5}
   394→              onValueChange={(v) => onChange(s.key, v[0])}
   395→              aria-label={s.label}
   396→            />
   397→          </div>
   398→        ))}
   399→      </div>
   400→
   401→      <p className="text-xs text-app-secondary leading-relaxed">
   402→        النسب بتتعدّل تلقائياً بحيث يكون مجموعها 100%.
   403→      </p>
   404→    </div>
   405→  );
   406→}

   421→function ComparisonTable({
   422→  scores,
   423→  solutions,
   424→  selectedIds,
   425→  onToggle,
   426→  expandedIds,
   427→  onToggleExpand,
   428→  causes,
   429→}: {

   433→  onToggle: (id: string) => void;

   457→            const sol = solutions.find((s) => s.id === sc.solutionId);

   459→            const selected = selectedIds.includes(sc.solutionId);
   460→            const disabled = !selected && selectedIds.length >= MAX_SELECTED;

   483→                      isExpanded && "border-b-app-primary/30",
   484→                    )}
   485→                  >
   486→                    {/* Checkbox cell */}
   487→                    <TableCell className="text-center">
   488→                      <Checkbox
   489→                        checked={selected}
   490→                        disabled={disabled}
   491→                        onCheckedChange={() => onToggle(sc.solutionId)}
   492→                        onClick={(e) => e.stopPropagation()}
   493→                        aria-label={`اختيار ${sol.label_ar}`}
   494→                      />
   495→                    </TableCell>
   496→
   497→                    {/* "تفاصيل" trigger button cell */}
   498→                    <TableCell className="text-center">
   499→                      <CollapsibleTrigger asChild>
   500→                        <button
   501→                          type="button"
   502→                          className="size-8 rounded-md inline-flex items-center justify-center text-app-secondary hover:text-app-primary hover:bg-app-primary/10 transition-colors"
   503→                          aria-label="تفاصيل الحل"
   504→                          aria-expanded={isExpanded}
   505→                          onClick={(e) => e.stopPropagation()}
   506→                        >
   507→                          <ChevronDown
   508→                            className={cn(
   509→                              "size-4 transition-transform duration-200",
   510→                              isExpanded && "rotate-180",
   511→                            )}
   512→                            aria-hidden="true"
   513→                          />
   514→                        </button>
   515→                      </CollapsibleTrigger>
   516→                    </TableCell>
   517→
   518→                    {/* Solution name cell — also a trigger (per task a) */}
   519→                    <TableCell className="text-app-text font-medium text-xs max-w-[180px]">
   520→                      <CollapsibleTrigger asChild>
   521→                        <button
   522→                          type="button"
   523→                          className="text-start flex items-center gap-1.5 hover:text-app-primary transition-colors"
   524→                          aria-label={`عرض تفاصيل ${sol.label_ar}`}
   525→                          aria-expanded={isExpanded}
   526→                          onClick={(e) => e.stopPropagation()}
   527→                        >
   528→                          {isTop && (
   529→                            <span className="inline-flex items-center gap-0.5 shrink-0 bg-amber-400 text-amber-950 rounded-full px-1.5 py-0.5 text-[10px] font-bold">
   530→                              <Crown className="size-2.5" aria-hidden="true" />
   531→                              الأفضل لك

   535→                        </button>
   536→                      </CollapsibleTrigger>

   886→// Lets the user pick exactly 2 solutions and see them compared
   887→// across all criteria with star ratings, winner highlights, and
   888→// a profile-based "which is better for me?" insight.

   891→function truncateLabel(s: string, max = 36): string {
   892→  if (s.length <= max) return s;
   893→  return `${s.slice(0, max).trimEnd()}…`;
   894→}
   895→
   896→/** Picks the winner side based on which value is "better". Returns null on ties. */
   897→function pickWinner(
   898→  a: number,
   899→  b: number,
   900→  mode: "higher" | "lower",
   901→): "left" | "right" | null {
   902→  if (a === b) return null;
   903→  if (mode === "higher") return a > b ? "left" : "right";
   904→  return a < b ? "left" : "right";
   905→}
   906→
   907→/** Star rating row — filled stars for `value`, empty for the rest (out of `max`). */
   908→function StarRating({ value, max = 5 }: { value: number; max?: number }) {
   909→  return (
   910→    <span
   911→      className="inline-flex items-center gap-1.5"
   912→      aria-label={`${value} من ${max} نجوم`}
   913→      role="img"
   914→    >
   915→      <span className="inline-flex items-center gap-0.5" aria-hidden="true">
   916→        {Array.from({ length: max }, (_, i) => (
   917→          <Star
   918→            key={i}
   919→            className={cn(
   920→              "size-3 sm:size-3.5 transition-colors",
   921→              i < value
   922→                ? "fill-amber-400 text-amber-400"
   923→                : "fill-transparent text-app-secondary/40",
   924→            )}
   925→          />
   926→        ))}
   927→      </span>
   928→      <span className="text-xs font-bold text-app-text tabular-nums">
   929→        {value}/{max}
   930→      </span>
   931→    </span>
   932→  );
   933→}

  1194→/**
  1195→ * Analyzes the two solutions against the user's profile and produces
  1196→ * the most relevant "which is better for me?" insight.
  1197→ *
  1198→ * Priority:
  1199→ * 1. If scores within 5 points → "try both"
  1200→ * 2. If one has higher fit → "X أنسب لملاءمتك"
  1201→ * 3. If urgent + one is faster → "X أسرع ومناسب لقرب الامتحان"
  1202→ * 4. If limited time + one needs less daily time → "X يحتاج وقت أقل يومياً"
  1203→ * 5. Default: higher score wins
  1204→ */
  1205→function analyzeFit(
  1206→  leftSol: Solution,
  1207→  rightSol: Solution,
  1208→  leftScore: SolutionScore,
  1209→  rightScore: SolutionScore,
  1210→  profile: Profile | null,
  1211→): string {
  1212→  const scoreDiff = Math.abs(leftScore.score - rightScore.score);
  1213→
  1214→  // 1. Very close scores → recommend trying both
  1215→  if (scoreDiff <= 5) {
  1216→    return "الحلين متقاربين جداً — جرّب الاتنين وشوف إيه اللي يناسبك أكتر.";
  1217→  }
  1218→
  1219→  // 2. Higher fit → recommend that one
  1220→  if (leftScore.fit !== rightScore.fit) {
  1221→    const fitWinner = leftScore.fit > rightScore.fit ? "الحل الأول" : "الحل الثاني";
  1222→    return `${fitWinner} أنسب لملاءمتك (وقتك/فريقك).`;
  1223→  }
  1224→
  1225→  // 3. Urgent + speed difference
  1226→  if (profile?.urgent && leftScore.S !== rightScore.S) {
  1227→    const faster = leftScore.S > rightScore.S ? "الحل الأول" : "الحل الثاني";
  1228→    return `${faster} أسرع ومناسب لقرب الامتحان.`;
  1229→  }
  1230→
  1231→  // 4. Limited time + daily minutes difference
  1232→  if (
  1233→    profile &&
  1234→    profile.minutesPerDay <= 30 &&
  1235→    leftSol.dailyMinutes !== rightSol.dailyMinutes
  1236→  ) {
  1237→    const lessTime =
  1238→      leftSol.dailyMinutes < rightSol.dailyMinutes ? "الحل الأول" : "الحل الثاني";
  1239→    return `${lessTime} يحتاج وقت أقل يومياً.`;
  1240→  }
  1241→
  1242→  // 5. Default: higher score
  1243→  const better = leftScore.score > rightScore.score ? "الحل الأول" : "الحل الثاني";
  1244→  const betterScore = leftScore.score > rightScore.score ? leftScore : rightScore;
  1245→  const worseScore = leftScore.score > rightScore.score ? rightScore : leftScore;
  1246→  return `${better} درجته أعلى (${betterScore.score} مقابل ${worseScore.score}).`;
  1247→}

  1284→  profile,
  1285→  onGoToScenarios,
  1286→}: {
  1287→  open: boolean;
  1288→  onOpenChange: (open: boolean) => void;
  1289→  scores: SolutionScore[];
  1290→  solutions: Solution[];
  1291→  causes: { id: string; label_ar: string }[];

  1297→  const [rightId, setRightId] = React.useState<string>("");

  1304→      setRightId(scores[1].solutionId);
  1305→    }
  1306→    wasOpenRef.current = open;
  1307→  }, [open, scores]);
  1308→

  1310→  const safeLeftId = scores.find((s) => s.solutionId === leftId)
  1311→    ? leftId
  1312→    : (scores[0]?.solutionId ?? "");

  1314→    scores.find((s) => s.solutionId === rightId) && rightId !== safeLeftId
  1315→      ? rightId

  1333→              />

  1337→          <p className="text-sm text-app-secondary">

  1347→      <DialogContent className="sm:max-w-3xl max-h-[92vh] overflow-y-auto p-4 sm:p-6">

  1351→              className="size-5 text-app-primary shrink-0"
  1352→              aria-hidden="true"
  1353→            />
  1354→            مقارنة حلوين
  1355→          </DialogTitle>
  1356→          <p className="text-xs text-app-secondary leading-relaxed">
  1357→            اختار حلّين وقارنهم على كل المعايير — الخلية الخضراء هي الأحسن في كل صف.
  1358→          </p>
  1359→        </DialogHeader>
  1360→
  1361→        {/* Solution pickers */}
  1362→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
  1363→          <SolutionPicker
  1364→            value={safeLeftId}
  1365→            onChange={setLeftId}
  1366→            scores={scores}
  1367→            solutions={solutions}
  1368→            excludeId={safeRightId}
  1369→            label="الحل الأول"
  1370→          />
  1371→          <SolutionPicker

  1392→          <h3 className="text-sm font-semibold text-app-text flex items-center gap-1.5">
